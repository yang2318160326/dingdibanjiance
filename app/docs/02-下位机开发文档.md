## 2. 核心数据结构

### 2.1 数据记录 (record.h)

```c
/* 传感器数据记录 - 32字节固定长度 */
typedef struct __attribute__((packed)) {
    uint32_t timestamp;         // Unix时间戳 (秒), 大端序
    uint8_t  sensor_addr;       // 传感器Modbus地址 (固定, 来自配置)
    uint8_t  status;            // 0=OK, 1=超时, 2=CRC错误, 3=异常
    uint16_t reg_count;         // 寄存器值数量
    uint16_t reg_values[8];     // 最多8个寄存器值 (16字节)
    uint32_t sequence_num;      // 单调递增序号
    uint16_t crc16;             // CRC16校验 (字节0x00~0x1B)
    uint16_t reserved;          // 保留
} SensorRecord_t;               // 32字节

#define RECORD_STATUS_OK            0x00
#define RECORD_STATUS_TIMEOUT       0x01
#define RECORD_STATUS_CRC_ERROR     0x02
#define RECORD_STATUS_MODBUS_EXC    0x03
```

### 2.2 系统配置 (config_manager.h) -- 简化版

```c
/* 数据类型定义 */
#define DATA_TYPE_UINT16    0x00
#define DATA_TYPE_INT16     0x01
#define DATA_TYPE_UINT32    0x02
#define DATA_TYPE_FLOAT32   0x03
#define DATA_TYPE_RAW       0x04

/* 系统配置 - 单传感器版本 */
typedef struct __attribute__((packed)) {
    uint32_t magic;                     // 魔数: 0x434F4E46
    uint32_t version;                   // 配置版本号
    uint32_t device_id;                 // 设备ID
    char     device_name[16];           // 设备名称

    /* 采样参数 */
    uint32_t sampling_interval_sec;     // 采样间隔 (秒), 0=连续

    /* 单传感器配置 */
    uint8_t  sensor_addr;               // Modbus从站地址 (1-247)
    uint16_t sensor_start_reg;          // 起始寄存器地址
    uint16_t sensor_reg_count;          // 读取数量 (1-125)
    uint8_t  sensor_data_type;          // 数据类型

    /* 通信参数 */
    uint32_t modbus_baudrate;           // 波特率 (默认9600)
    uint8_t  modbus_parity;             // 0=None, 1=Odd, 2=Even

    uint16_t crc16;                     // 配置校验
} SystemConfig_t;                       // 约40字节
```

### 2.3 存储元数据 (data_store.h)

```c
typedef struct __attribute__((packed)) {
    uint32_t magic;             // 0x44415441

    uint32_t total_records;     // 有效记录数
```

## 3. 模块详细设计

### 3.1 Modbus RTU主站 (modbus_master.h)

#### API定义

```c
/**
 * @brief 初始化Modbus主站
 */
int Modbus_Init(uint32_t baudrate, uint8_t parity);

/**
 * @brief 读取保持寄存器 (功能码0x03)
 * @param slave_addr    从站地址 (来自配置)
 * @param start_reg     起始寄存器 (来自配置)
 * @param count         读取数量 (来自配置)
 * @param out_values    输出缓冲区
 * @param timeout_ms    超时时间
 * @return >0=成功(寄存器数), <0=错误码
 */
int Modbus_ReadHoldingRegisters(uint8_t slave_addr, uint16_t start_reg,
                                 uint16_t count, uint16_t *out_values,
                                 uint32_t timeout_ms);

/**
 * @brief 读取输入寄存器 (功能码0x04)
 */
int Modbus_ReadInputRegisters(uint8_t slave_addr, uint16_t start_reg,
                               uint16_t count, uint16_t *out_values,
                               uint32_t timeout_ms);

/**
 * @brief 计算Modbus CRC16
 */
uint16_t Modbus_CRC16(const uint8_t *data, uint16_t len);
```

### 3.2 传感器管理器 (sensor_manager.h) -- 简化版

单传感器采集, 无需多传感器轮询状态机, 流程大幅简化:

```c
/**
 * @brief 初始化传感器管理器
 * @param config  系统配置指针 (包含单个传感器参数)
 */
int SensorManager_Init(const SystemConfig_t *config);

/**
 * @brief 执行一次传感器采集
 *        读取单个传感器, 打包成记录, 存储到Flash
 * @return 0=成功, <0=错误码
 */
int SensorManager_AcquireOnce(void);
```

#### 简化后的采集流程

```
┌──────────────┐
│ 读取传感器   │  Modbus_ReadHoldingRegisters(
│ (单个)       │    config->sensor_addr,
│              │    config->sensor_start_reg,
│              │    config->sensor_reg_count, ...)
└──────┬───────┘
       │
       ▼
┌──────────────┐
│ 打包记录     │  填充SensorRecord_t
│              │  - timestamp (RTC)
│              │  - sensor_addr (配置)
│              │  - status (读取结果)
│              │  - reg_values[] (原始值)
│              │  - sequence_num (递增)
└──────┬───────┘
       │
       ▼
┌──────────────┐
│ 存储到Flash  │  DataStore_Append(&record)
└──────────────┘
```

## 4. 主循环设计 (main.c)

```c
int main(void)
{
    /* 1. HAL初始化 */
    HAL_Init();
    SystemClock_Config();

    /* 2. BSP初始化 */
    BSP_RS485_Init();
    BSP_BLE_Init();
    BSP_LED_Init();

    /* 3. 应用层初始化 */
    ConfigManager_Init();
    DataStore_Init();
    RtcScheduler_Init();

    const SystemConfig_t *config = ConfigManager_GetConfig();
    Modbus_Init(config->modbus_baudrate, config->modbus_parity);
    SensorManager_Init(config);
    BleTransport_Init();
    CmdDispatcher_Init();
    PowerManager_Init();

    /* 4. 检查唤醒原因 */
    if (PowerManager_IsWakeupFromSleep()) {
        goto DO_ACQUIRE;
    }

    /* 5. 主循环 */
    while (1) {
DO_ACQUIRE:
        /* 执行一次传感器采集 (单传感器, 很快) */
        SensorManager_AcquireOnce();

        /* 检查BLE是否有数据 */
        if (BleTransport_HasPendingData()) {
            uint8_t cmd, payload[256];
            uint16_t len;
            if (BleTransport_ReceiveFrame(&cmd, payload, &len, 100) == 0) {
                CmdDispatcher_Handle(cmd, payload, len);
            }
        } else if (config->sampling_interval_sec > 0) {
            /* 非连续模式 -> 进入睡眠 */
            RtcScheduler_SetNextAlarm(config->sampling_interval_sec);
            PowerManager_EnterSleep(config->sampling_interval_sec);
        }

        HAL_Delay(10);
    }
}
```

## 5. 低功耗设计

| 模式 | 电流 | 占比 | 说明 |
|------|------|------|------|
| STOP2 + RTC | ~5μA | 99%+ | 绝大部分时间 |
| 运行模式 (采集) | ~10mA | <1% | 单传感器采集约50ms |
| BLE活动 | ~15mA | 极少 | 仅APP连接时 |

## 6. 错误处理

| 错误类型 | 处理方式 |
|----------|----------|
| Modbus超时 | 记录status=TIMEOUT, 仍然存储 |
| Modbus CRC错误 | 记录status=CRC_ERROR, 重试1次 |
| Flash写入失败 | 跳过本次, LED告警 |
| Flash满 | 覆盖最旧记录 |
| BLE超时 | 丢弃帧, 等待下一帧 |
| 配置CRC错误 | 恢复默认配置 |
